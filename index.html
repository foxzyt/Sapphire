Syntax

Elegant by design.

Explore Sapphire's clean, expressive syntax across multiple domains.

basic_syntax.sp 
// Variables & type inference
var name  = "Sapphire";
var value = 42;
var pi    = 3.14159;
var flag  = true;

// f-string interpolation
print(f"Language: {name}, Value: {value}");

// Nullish coalescing (??)
var input    = nil;
var username = input ?? "Guest";
print(f"Welcome, {username}!");

// Optional chaining (?.)  
var guild = player?.guild?.name ?? "None";
// Named functions
function calculateArea(w, h) {
    return w * h;
}

// Concise arrow functions
var square  = (x) => x * x;
var greet   = (n) => print(f"Hi, {n}!");

// Recursion (factorial)
function factorial(n) {
    if (n == 0) return 1;
    return n * factorial(n - 1);
}

print(f"10! = {factorial(10)}");
greet("Sapphire User");
class Animal {
    function speak() {
        print("...");
    }
}

class Dog extends Animal {
    function init(name) {
        this.name = name;
    }
    function speak() {
        print(f"{this.name} says: Woof!");
    }
}

var buddy = Dog("Buddy");
buddy.speak(); // Buddy says: Woof!
import "http";

// Async function with await
async function fetchData(url) {
    var res = await http_get(url);
    print(f"Status: {res.status}");
    return res.body;
}

// Parallel async calls
async function main() {
    var data = await fetchData(
        "https://api.example.com/users"
    );
    print(data);
}
main();
// Native Desktop UI — no frameworks needed!
var count = 0;

var onIncrement = () => {
    count = count + 1;
    print(f"Count: {count}");
};

var app = Window({
    "title": "Counter",
    "width": 400, "height": 300,
    "children": [
        Text({ "text": "Sapphire Counter!" }),
        Button({
            "label": "Increment",
            "onClick": onIncrement
        })
    ]
});
while (updateUI()) Render(app);

Quick Start

Up and running in minutes.

Three simple steps to write and run your first Sapphire program.

1

Download the Compiler

Head to our GitHub Releases page and download the latest sapphire.exe for Windows. No installation required — just drop it anywhere.

2

Write Your Code

Create a file named main.sp in any text editor. Write your first lines of Sapphire — the syntax will feel immediately familiar.

3

Run & Ship

Execute with sapphire.exe main.sp. When ready to ship, pack into a self-contained binary with the built-in packaging tool.
